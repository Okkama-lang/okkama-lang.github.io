Linux: install from the archive

Run these commands in a fresh directory. Check that the checksum reports OK before extracting. The installer copies into ~/.local and replaces any Okkama installation already at that prefix.

Linux x86_64 · terminal
curl -fLO https://github.com/Okkama-lang/okkama/releases/download/v0.6.3/okkama-0.6.3-linux-x86_64.tar.gz
curl -fLO https://github.com/Okkama-lang/okkama/releases/download/v0.6.3/okkama-0.6.3-linux-x86_64.tar.gz.sha256
sha256sum -c okkama-0.6.3-linux-x86_64.tar.gz.sha256
tar -xzf okkama-0.6.3-linux-x86_64.tar.gz
./okkama-0.6.3-linux-x86_64/install.sh --prefix "$HOME/.local"
export PATH="$HOME/.local/bin:$PATH"
okkama --version

The export applies to this terminal. Add it to your shell configuration if ~/.local/bin is not already on your PATH. After replacing an older install, open a new terminal or run hash -r in Bash.

module quickstart/payment v1.0.0;

import "io/stdout" take Stdout;

role Created {
    status() is string;
    authorize() is ^Authorized;
}

role Authorized {
    status() is string;
    capture() is ^Captured;
}

role Captured {
    status() is string;
}

hidden enum Stage = (created, authorized, captured);
enum PaymentError is Error = (invalid_transition);

hidden object Payment does Created, Authorized, Captured {
    has stage is Stage = Stage.created;
    Payment();

    status() is string {
        match stage {
            case Stage.created { "created"; }
            case Stage.authorized { "authorized"; }
            case Stage.captured { "captured"; }
        }
    }

    authorize() is ^Payment {
        confirm stage == Stage.created else throw PaymentError.invalid_transition;
        ^stage = Stage.authorized;
    }

    capture() is ^Payment {
        confirm stage == Stage.authorized else throw PaymentError.invalid_transition;
        ^stage = Stage.captured;
    }
}

object Payments {
    Payments();
    create() is Created { Payment(); }
}

main() is int {
    let out = Stdout();
    let created = Payments().create();
    let authorized = created.authorize();
    let captured = authorized.capture();
    out.writeln("Original: " + created.status());
    out.writeln("After authorization: " + authorized.status());
    out.writeln("After capture: " + captured.status());
    0;
}
This is a model, not a payment gateway.No network request is made and no money moves. It demonstrates local type contracts and immutable values, not database locking, idempotency, or protection against concurrent charges.

03 /

Let the compiler say no.

Now ask a newly created payment to capture immediately. In payment.ok, replace this line:

Valid transition
let captured = authorized.capture();

with:

Invalid transition
let captured = created.capture();

Save the changed file as payment-invalid.ok, or download the prepared version. Check it without running:

Download payment-invalid.ok
Compile-time check
okkama check payment-invalid.ok
Expected compiler diagnostic · exit status 1
payment-invalid.ok:54:28: semantic error: Role "Created" has no action "capture"

The compiler rejects this before execution. The Created role offers authorize(), not capture(). The full object is hidden behind that role, so callers follow the intended protocol.

THE POINT

The next valid action is in the type, not just in a comment. That is the combination of objects, roles, and successors.
